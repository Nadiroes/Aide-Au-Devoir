   351→export interface RevisionSectionProps {
   352→  documents: DocumentDTO[];
   353→  onDocumentsChanged: () => void;
   354→  selectedDocumentId: string | null;
   355→  onSelectDocument: (id: string) => void;
   356→  greetDocumentId?: string | null;
   357→  onGreetConsumed?: () => void;
   358→  onOpenCourses?: () => void;
   359→  onMemoryChanged?: () => void;
   360→}
   361→
   362→export function RevisionSection({
   363→  documents,
   364→  onDocumentsChanged,
   365→  selectedDocumentId,
   366→  onSelectDocument,
   367→  greetDocumentId,
   368→  onGreetConsumed,
   369→  onOpenCourses,
   370→  onMemoryChanged,
   371→}: RevisionSectionProps) {
   372→  const docId = selectedDocumentId;
   373→  const currentDoc = useMemo(
   374→    () => documents.find((d) => d.id === docId) ?? null,
   375→    [documents, docId]
   376→  );
   377→
   378→  const [folders, setFolders] = useState<FolderDTO[]>([]);
   379→  const [detail, setDetail] = useState<DocumentDetailDTO | null>(null);
   380→  const [detailLoading, setDetailLoading] = useState(false);
   381→  const [qas, setQas] = useState<RevisionQADTO[]>([]);
   382→  const [qasLoading, setQasLoading] = useState(false);
   383→  const [filter, setFilter] = useState<QAFilter>("all");
   384→  const [sort, setSort] = useState<QASort>("recent");
   385→  const [deletingQaId, setDeletingQaId] = useState<string | null>(null);
   386→
   387→  const [busy, setBusy] = useState(false);
   388→  const [agents, setAgents] = useState<AgentDisplay[]>(freshAgents);
   389→  const [live, setLive] = useState<LiveExchange | null>(null);
   390→  const [liveRating, setLiveRating] = useState<QARating | null>(null);
   391→  const [answerInput, setAnswerInput] = useState("");
   392→  const [dismissedPrompt, setDismissedPrompt] = useState(false);
   393→
   394→  // Audio reader preferences (persisted in Paramètres).
   395→  const [voice, setVoice] = useState<string>("auto");
   396→  const [speed, setSpeed] = useState<number>(1);
   397→
   398→  const player = useTtsPlayer();
   399→  const abortRef = useRef<AbortController | null>(null);
   400→  const streamRef = useRef<HTMLDivElement | null>(null);
   401→  const detailRef = useRef<DocumentDetailDTO | null>(null);
   402→  const liveRef = useRef<LiveExchange | null>(null);
   403→
   404→  const recorder = useRecorder((text) =>
   405→    setAnswerInput((prev) => (prev.trim().length > 0 ? `${prev.trim()} ${text}` : text))
   406→  );
   407→
   408→  useEffect(() => {
   409→    detailRef.current = detail;
   410→  }, [detail]);
   411→
   412→  // Mirror of the live exchange for use inside the SSE handler.
   413→  useEffect(() => {
   414→    liveRef.current = live;
   415→  }, [live]);
   416→
   417→  // Toast dictation errors.
   418→  useEffect(() => {
   419→    if (recorder.error) toast.error(recorder.error);
   420→  }, [recorder.error]);
   421→
   422→  // Load persisted TTS preferences once.
   423→  useEffect(() => {
   424→    const prefs = getTtsPrefs();
   425→    setVoice(prefs.voice ?? "auto");
   426→    setSpeed(prefs.speed ?? 1);
   427→  }, []);
   428→
   429→  // Non-critical: folder list for the grouped selector.
   430→  useEffect(() => {
   431→    void (async () => {
   432→      try {
   433→        setFolders(await fetchFolders());
   434→      } catch {
   435→        // Folders are optional — ignore silently.
   436→      }
   437→    })();
   438→  }, []);
   439→
   440→  // ---------- Data loading ----------
   441→
   442→  const loadDetail = useCallback(async (id: string) => {
   443→    setDetailLoading(true);
   444→    try {
   445→      setDetail(await fetchDocumentDetail(id));
   446→    } catch (err) {
   447→      toast.error(err instanceof Error ? err.message : "Impossible de charger le cours");
   448→    } finally {
   449→      setDetailLoading(false);
   450→    }
   451→  }, []);
   452→
   453→  const loadQas = useCallback(
   454→    async (id: string) => {
   455→      setQasLoading(true);
   456→      try {
   457→        setQas(await fetchQAs(id, { filter, sort }));
   458→      } catch (err) {
   459→        toast.error(err instanceof Error ? err.message : "Impossible de charger la fiche de révision");
   460→      } finally {
   461→        setQasLoading(false);
   462→      }
   463→    },
   464→    [filter, sort]
   465→  );
   466→
   467→  // Document switch — hard reset of the workspace (stops audio too).
   468→  useEffect(() => {
   469→    player.stop();
   470→    setLive(null);
   471→    setLiveRating(null);
   472→    setDismissedPrompt(false);
   473→    setAgents(freshAgents());
   474→    setBusy(false);
   475→    abortRef.current?.abort();
   476→    abortRef.current = null;
   477→    setDetail(null);
   478→    setQas([]);
   479→    setAnswerInput("");
   480→    if (docId) {
   481→      void loadDetail(docId);
   482→      void loadQas(docId);
   483→    }
   484→  }, [docId]);
   485→
   486→  // Refetch when the filter / sort changes (same document).
   487→  useEffect(() => {
   488→    if (docId) void loadQas(docId);
   489→  }, [filter, sort]);
   490→
   491→  // Auto-scroll the Q&A stream while busy.
   492→  useEffect(() => {
   493→    if (busy && streamRef.current) {
   494→      streamRef.current.scrollTop = streamRef.current.scrollHeight;
   495→    }
   496→  }, [live?.coach, live?.feedback, busy]);
   497→
   498→  // ---------- SSE state machine ----------
   499→
   500→  const patchAgent = useCallback((id: string, patch: Partial<AgentDisplay>) => {
   501→    setAgents((prev) => prev.map((a) => (a.id === id ? { ...a, ...patch } : a)));
   502→  }, []);
   503→
   504→  const handleEvent = useCallback(
   505→    (event: RevisionEvent) => {
   506→      switch (event.type) {
   507→        case "revision.start":
   508→          break;
   509→        case "agent.start":
   510→          patchAgent(event.id, { status: "running", logs: [], summary: undefined, content: undefined });
   511→          break;
   512→        case "agent.log":
   513→          setAgents((prev) =>
   514→            prev.map((a) => (a.id === event.id ? { ...a, logs: [...a.logs, event.message] } : a))
   515→          );
   516→          break;
   517→        case "agent.done": {
   518→          patchAgent(event.id, { status: "done", summary: event.summary, content: event.content });
   519→          // Optional auto-read: when the coach finishes an open question, speak it aloud.
   520→          if (event.id === "coach" && getTtsPrefs().autoread && detailRef.current) {
   521→            const current = liveRef.current;
   522→            if (current && current.mode === "question" && current.userText === null && current.coach.trim().length > 0) {
   523→              player.play(stripMd(current.coach), { language: detailRef.current.language });
   524→            }
   525→          }
   526→          break;
   527→        }
   528→        case "agent.error":
   529→          patchAgent(event.id, { status: "error", summary: event.message });
   530→          toast.error(event.message);
   531→          break;
   532→        case "message.token":
   533→          setLive((prev) => {
   534→            const base: LiveExchange =
   535→              prev ?? { mode: "question", coach: "", userText: null, feedback: "", evaluation: null, qaId: null };
   536→            // Once the user's answer is in, tokens are evaluator feedback.
   537→            if (base.mode === "question" && base.userText !== null) {
   538→              return { ...base, feedback: base.feedback + event.token };
   539→            }
   540→            return { ...base, coach: base.coach + event.token };
   541→          });
   542→          break;
   543→        case "evaluation":
   544→          setLive((prev) => {
   545→            if (!prev) return prev;
   546→            return {
   547→              ...prev,
   548→              evaluation: {
   549→                verdict: event.verdict,
   550→                feedback: event.feedback,
   551→                modelAnswer: event.modelAnswer,
   552→                score: event.score,
   553→              },
   554→              feedback: prev.feedback.trim().length > 0 ? prev.feedback : event.feedback,
   555→            };
   556→          });
   557→          break;
   558→        case "qa.saved": {
   559→          const qa = event.qa;
   560→          const isQuestionOnly = !qa.answer || qa.answer.trim().length === 0;
   561→          if (isQuestionOnly) {
   562→            setLive((prev) => {
   563→              const base: LiveExchange =
   564→                prev ?? { mode: "question", coach: "", userText: null, feedback: "", evaluation: null, qaId: null };
   565→              return {
   566→                ...base,
   567→                qaId: qa.id,
   568→                coach: base.coach.trim().length > 0 ? base.coach : qa.question,
   569→              };
   570→            });
   571→            // Auto-read preference: the coach question is spoken aloud.
   572→            if (getTtsPrefs().autoread && detailRef.current) {
   573→              player.play(stripMd(qa.question), { language: detailRef.current.language });
   574→            }
   575→          } else {
   576→            setLive((prev) => (prev ? { ...prev, qaId: prev.qaId ?? qa.id } : prev));
   577→            setQas((prev) => {
   578→              const idx = prev.findIndex((q) => q.id === qa.id);
   579→              if (idx === -1) return [qa, ...prev];
   580→              const copy = [...prev];
   581→              copy[idx] = qa;
