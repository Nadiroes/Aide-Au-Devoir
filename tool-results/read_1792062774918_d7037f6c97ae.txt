   219→   219→ingestion.py — PDF / DOCX / TXT / IMG
   220→   220→python
   221→   221→from pathlib import Path
   222→   222→import pdfplumber, pytesseract
   223→   223→from docx import Document as DocxDocument
   224→   224→from PIL import Image
   225→   225→
   226→   226→def extract_text(path: Path) -> str:
   227→   227→    ext = path.suffix.lower()
   228→   228→    if ext == ".pdf":
   229→   229→        with pdfplumber.open(path) as pdf:
   230→   230→            txt = "\n".join(p.extract_text() or "" for p in pdf.pages)
   231→   231→        # Si PDF scanné → OCR fallback
   232→   232→        if len(txt.strip()) < 50:
   233→   233→            imgs = []
   234→   234→            with pdfplumber.open(path) as pdf:
   235→   235→                for p in pdf.pages:
   236→   236→                    imgs.append(p.to_image(resolution=200).original)
   237→   237→            return "\n".join(pytesseract.image_to_string(i, lang="fra+eng") for i in imgs)
   238→   238→        return txt
   239→   239→    if ext == ".docx":
   240→   240→        return "\n".join(p.text for p in DocxDocument(path).paragraphs)
   241→   241→    if ext == ".txt":
   242→   242→        return path.read_text(encoding="utf-8", errors="ignore")
   243→   243→    if ext in (".png", ".jpg", ".jpeg"):
   244→   244→        return pytesseract.image_to_string(Image.open(path), lang="fra+eng")
   245→   245→    raise ValueError(f"Format non supporté : {ext}")
   246→   246→
   247→   247→def chunk(text: str, size: int = 800, overlap: int = 120):
   248→   248→    chunks, i = [], 0
   249→   249→    while i < len(text):
   250→   250→        chunks.append(text[i:i+size])
   251→   251→        i += size - overlap
   252→   252→    return [c for c in chunks if c.strip()]

   274→   274→        out.append({"text": doc, "meta": meta, "score": 1 - dist})
   275→   275→    return out

   951→   951→backend/events.py
   952→   952→python
   953→   953→import asyncio
   954→   954→from typing import AsyncIterator, Callable, Any
   955→   955→
   956→   956→class EventBus:
   957→   957→    """Bus d'événements asyncio pour pousser des messages SSE depuis les agents."""
   958→   958→    def __init__(self):
   959→   959→        self._queue: asyncio.Queue[dict] = asyncio.Queue()
   960→   960→        self._closed = False
   961→   961→
   962→   962→    async def emit(self, event: str, data: dict):
   963→   963→        if self._closed:
   964→   964→            return
   965→   965→        await self._queue.put({"event": event, "data": data})
   966→   966→
   967→   967→    async def close(self):
   968→   968→        if self._closed:
   969→   969→            return
   970→   970→        self._closed = True
   971→   971→        await self._queue.put({"event": "__end__", "data": {}})
   972→   972→
   973→   973→    async def stream(self) -> AsyncIterator[dict]:
   974→   974→        while True:
   975→   975→            ev = await self._queue.get()
   976→   976→            if ev["event"] == "__end__":
   977→   977→                break
   978→   978→            yield ev
   979→   979→
   980→   980→    @staticmethod
   981→   981→    def format_sse(event: str, data: dict) -> str:
   982→   982→        import json
   983→   983→        return f"event: {event}\ndata: {json.dumps(data, ensure_ascii=False)}\n\n"

  1088→  1088→async def run_pipeline_stream(question: str, provider: str) -> AsyncIterator[dict]:
  1089→  1089→    """Pipeline complet avec streaming d'événements SSE."""
  1090→  1090→    bus = EventBus()
  1091→  1091→
  1092→  1092→    async def worker():
  1093→  1093→        try:
  1094→  1094→            await bus.emit("pipeline.start", {"question": question, "provider": provider})
  1095→  1095→
  1096→  1096→            plan = await PlannerAgent(provider, bus).run(question)
  1097→  1097→
  1098→  1098→            hits = await RetrieverAgent(provider, bus).run(question)
  1099→  1099→            context = "\n---\n".join(h["text"] for h in hits) or "(aucun cours pertinent)"
  1100→  1100→
  1101→  1101→            answer = await SolverAgent(provider, bus).run(question, context, plan)
  1102→  1102→
  1103→  1103→            await TutorAgent(provider, bus).run(question, answer)
  1104→  1104→
  1105→  1105→            await bus.emit("pipeline.done", {"answer": answer})
  1106→  1106→        except Exception as e:
  1107→  1107→            await bus.emit("pipeline.error", {"message": str(e)})
  1108→  1108→        finally:
  1109→  1109→            await bus.close()
  1110→  1110→
  1111→  1111→    task = asyncio.create_task(worker())
  1112→  1112→    async for ev in bus.stream():
  1113→  1113→        yield ev
  1114→  1114→    await task

  1475→  1475→export type SseHandler = (event: string, data: any) => void;
  1476→  1476→
  1477→  1477→export async function sseStream(
  1478→  1478→  url: string,
  1479→  1479→  body: any,
  1480→  1480→  onEvent: SseHandler,
  1481→  1481→  signal?: AbortSignal,
  1482→  1482→) {
  1483→  1483→  const res = await fetch(url, {
  1484→  1484→    method: "POST",
  1485→  1485→    headers: { "Content-Type": "application/json", Accept: "text/event-stream" },
  1486→  1486→    body: JSON.stringify(body),
  1487→  1487→    signal,
  1488→  1488→  });
  1489→  1489→  if (!res.ok || !res.body) throw new Error(`SSE HTTP ${res.status}`);
  1490→  1490→
  1491→  1491→  const reader = res.body.getReader();
  1492→  1492→  const decoder = new TextDecoder("utf-8");
  1493→  1493→  let buffer = "";
  1494→  1494→
  1495→  1495→  while (true) {
  1496→  1496→    const { value, done } = await reader.read();
  1497→  1497→    if (done) break;
  1498→  1498→    buffer += decoder.decode(value, { stream: true });
  1499→  1499→
  1500→  1500→    let idx: number;
  1501→  1501→    while ((idx = buffer.indexOf("\n\n")) !== -1) {
  1502→  1502→      const raw = buffer.slice(0, idx);
  1503→  1503→      buffer = buffer.slice(idx + 2);
  1504→  1504→
  1505→  1505→      let event = "message";
  1506→  1506→      const dataLines: string[] = [];
  1507→  1507→      for (const line of raw.split("\n")) {
  1508→  1508→        if (line.startsWith(":")) continue;            // keep-alive
  1509→  1509→        if (line.startsWith("event:")) event = line.slice(6).trim();
  1510→  1510→        else if (line.startsWith("data:")) dataLines.push(line.slice(5).trim());
  1511→  1511→      }
  1512→  1512→      if (!dataLines.length) continue;
  1513→  1513→      try {
  1514→  1514→        onEvent(event, JSON.parse(dataLines.join("\n")));
  1515→  1515→      } catch {
  1516→  1516→        // ignore
  1517→  1517→      }
  1518→  1518→    }
  1519→  1519→  }
