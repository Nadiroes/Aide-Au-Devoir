   362→   362→export function RevisionSection({
   363→   363→  documents,
   364→   364→  onDocumentsChanged,
   365→   365→  selectedDocumentId,
   366→   366→  onSelectDocument,
   367→   367→  greetDocumentId,
   368→   368→  onGreetConsumed,
   369→   369→  onOpenCourses,
   370→   370→  onMemoryChanged,
   371→   371→}: RevisionSectionProps) {
   372→   372→  const docId = selectedDocumentId;
   373→   373→  const currentDoc = useMemo(
   374→   374→    () => documents.find((d) => d.id === docId) ?? null,
   375→   375→    [documents, docId]
   376→   376→  );
   377→   377→
   378→   378→  const [folders, setFolders] = useState<FolderDTO[]>([]);
   379→   379→  const [detail, setDetail] = useState<DocumentDetailDTO | null>(null);
   380→   380→  const [detailLoading, setDetailLoading] = useState(false);
   381→   381→  const [qas, setQas] = useState<RevisionQADTO[]>([]);
   382→   382→  const [qasLoading, setQasLoading] = useState(false);
   383→   383→  const [filter, setFilter] = useState<QAFilter>("all");
   384→   384→  const [sort, setSort] = useState<QASort>("recent");
   385→   385→  const [deletingQaId, setDeletingQaId] = useState<string | null>(null);
   386→   386→
   387→   387→  const [busy, setBusy] = useState(false);
   388→   388→  const [agents, setAgents] = useState<AgentDisplay[]>(freshAgents);
   389→   389→  const [live, setLive] = useState<LiveExchange | null>(null);
   390→   390→  const [liveRating, setLiveRating] = useState<QARating | null>(null);
   391→   391→  const [answerInput, setAnswerInput] = useState("");
   392→   392→  const [dismissedPrompt, setDismissedPrompt] = useState(false);
   393→   393→
   394→   394→  // Audio reader preferences (persisted in Paramètres).
   395→   395→  const [voice, setVoice] = useState<string>("auto");
   396→   396→  const [speed, setSpeed] = useState<number>(1);
   397→   397→
   398→   398→  const player = useTtsPlayer();
   399→   399→  const abortRef = useRef<AbortController | null>(null);
   400→   400→  const streamRef = useRef<HTMLDivElement | null>(null);
   401→   401→  const detailRef = useRef<DocumentDetailDTO | null>(null);
   402→   402→  const liveRef = useRef<LiveExchange | null>(null);
   403→   403→
   404→   404→  const recorder = useRecorder((text) =>
   405→   405→    setAnswerInput((prev) => (prev.trim().length > 0 ? `${prev.trim()} ${text}` : text))
   406→   406→  );
   407→   407→
   408→   408→  useEffect(() => {
   409→   409→    detailRef.current = detail;
   410→   410→  }, [detail]);
   411→   411→
   412→   412→  // Mirror of the live exchange for use inside the SSE handler.
   413→   413→  useEffect(() => {
   414→   414→    liveRef.current = live;
   415→   415→  }, [live]);
   416→   416→
   417→   417→  // Toast dictation errors.
   418→   418→  useEffect(() => {
   419→   419→    if (recorder.error) toast.error(recorder.error);
   420→   420→  }, [recorder.error]);
   421→   421→
   422→   422→  // Load persisted TTS preferences once.
   423→   423→  useEffect(() => {
   424→   424→    const prefs = getTtsPrefs();
   425→   425→    setVoice(prefs.voice ?? "auto");
   426→   426→    setSpeed(prefs.speed ?? 1);
   427→   427→  }, []);
   428→   428→
   429→   429→  // Non-critical: folder list for the grouped selector.
   430→   430→  useEffect(() => {
   431→   431→    void (async () => {
   432→   432→      try {
   433→   433→        setFolders(await fetchFolders());
   434→   434→      } catch {
   435→   435→        // Folders are optional — ignore silently.
   436→   436→      }
   437→   437→    })();
   438→   438→  }, []);
   439→   439→
   440→   440→  // ---------- Data loading ----------
   441→   441→
   442→   442→  const loadDetail = useCallback(async (id: string) => {
   443→   443→    setDetailLoading(true);
   444→   444→    try {
   445→   445→      setDetail(await fetchDocumentDetail(id));
   446→   446→    } catch (err) {
   447→   447→      toast.error(err instanceof Error ? err.message : "Impossible de charger le cours");
   448→   448→    } finally {
   449→   449→      setDetailLoading(false);
   450→   450→    }
   451→   451→  }, []);
   452→   452→
   453→   453→  const loadQas = useCallback(
   454→   454→    async (id: string) => {
   455→   455→      setQasLoading(true);
   456→   456→      try {
   457→   457→        setQas(await fetchQAs(id, { filter, sort }));
   458→   458→      } catch (err) {
   459→   459→        toast.error(err instanceof Error ? err.message : "Impossible de charger la fiche de révision");
   460→   460→      } finally {
   461→   461→        setQasLoading(false);
   462→   462→      }
   463→   463→    },
   464→   464→    [filter, sort]
   465→   465→  );
   466→   466→
   467→   467→  // Document switch — hard reset of the workspace (stops audio too).
   468→   468→  useEffect(() => {
   469→   469→    player.stop();
   470→   470→    setLive(null);
   471→   471→    setLiveRating(null);
   472→   472→    setDismissedPrompt(false);
   473→   473→    setAgents(freshAgents());
   474→   474→    setBusy(false);
   475→   475→    abortRef.current?.abort();
   476→   476→    abortRef.current = null;
   477→   477→    setDetail(null);
   478→   478→    setQas([]);
   479→   479→    setAnswerInput("");
   480→   480→    if (docId) {
   481→   481→      void loadDetail(docId);
   482→   482→      void loadQas(docId);
   483→   483→    }
   484→   484→  }, [docId]);
   485→   485→
