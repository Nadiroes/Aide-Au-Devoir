import base64, hashlib
from cryptography.fernet import Fernet
from config import MASTER_KEY

def _fernet() -> Fernet:
    key = base64.urlsafe_b64encode(hashlib.sha256(MASTER_KEY.encode()).digest())
    return Fernet(key)

def encrypt(value: str) -> str:
    return _fernet().encrypt(value.encode()).decode()

def decrypt(token: str) -> str:
    return _fernet().decrypt(token.encode()).decode()

def mask(key: str) -> str:
    return key[:4] + "…" + key[-4:] if len(key) > 8 else "****"

ingestion.py — PDF / DOCX / TXT / IMG

from pathlib import Path
import pdfplumber, pytesseract
from docx import Document as DocxDocument
from PIL import Image

def extract_text(path: Path) -> str:
    ext = path.suffix.lower()
    if ext == ".pdf":
        with pdfplumber.open(path) as pdf:
            txt = "\n".join(p.extract_text() or "" for p in pdf.pages)
        # Si PDF scanné → OCR fallback
        if len(txt.strip()) < 50:
            imgs = []
            with pdfplumber.open(path) as pdf:
                for p in pdf.pages:
                    imgs.append(p.to_image(resolution=200).original)
            return "\n".join(pytesseract.image_to_string(i, lang="fra+eng") for i in imgs)
        return txt
    if ext == ".docx":
        return "\n".join(p.text for p in DocxDocument(path).paragraphs)
    if ext == ".txt":
        return path.read_text(encoding="utf-8", errors="ignore")
    if ext in (".png", ".jpg", ".jpeg"):
        return pytesseract.image_to_string(Image.open(path), lang="fra+eng")
    raise ValueError(f"Format non supporté : {ext}")

def chunk(text: str, size: int = 800, overlap: int = 120):
    chunks, i = [], 0
    while i < len(text):
        chunks.append(text[i:i+size])
        i += size - overlap
    return [c for c in chunks if c.strip()]

import chromadb
from chromadb.utils.embedding_functions import SentenceTransformerEmbeddingFunction
from config import CHROMA_DIR

_client = chromadb.PersistentClient(path=str(CHROMA_DIR))
_embed = SentenceTransformerEmbeddingFunction(model_name="all-MiniLM-L6-v2")
_collection = _client.get_or_create_collection("cours", embedding_function=_embed)

def index_chunks(doc_id: int, title: str, chunks: list[str]):
    _collection.add(
        ids=[f"{doc_id}-{i}" for i in range(len(chunks))],
        documents=chunks,
        metadatas=[{"doc_id": doc_id, "title": title, "chunk": i} for i in range(len(chunks))],
    )

def search(query: str, k: int = 5):
    r = _collection.query(query_texts=[query], n_results=k)
    out = []
    for doc, meta, dist in zip(r["documents"][0], r["metadatas"][0], r["distances"][0]):
        out.append({"text": doc, "meta": meta, "score": 1 - dist})
    return out

PROVIDERS = {
    "mistral": {
        "base_url": "https://api.mistral.ai/v1",
        "model": "mistral-large-latest",
        "test_path": "/models",
    },
    "agnes": {
        # Remplacer par l'URL réelle du service Agnes
        "base_url": "https://api.agnes.ai/v1",
        "model": "agnes-chat",
        "test_path": "/models",
    },
}

def get_key(provider: str) -> str | None:
    db = SessionLocal()
    row = db.query(ApiKey).filter_by(provider=provider).first()
    db.close()
    return decrypt(row.encrypted) if row else None

async def verify_key(provider: str, key: str) -> bool:
    cfg = PROVIDERS[provider]
    async with httpx.AsyncClient(timeout=15) as c:
        r = await c.get(cfg["base_url"] + cfg["test_path"],
                        headers={"Authorization": f"Bearer {key}"})
    return r.status_code == 200

async def chat(provider: str, messages: list[dict], temperature=0.3) -> str:
    key = get_key(provider)
    if not key:
        raise RuntimeError(f"Clé {provider} non configurée")
    cfg = PROVIDERS[provider]
    async with httpx.AsyncClient(timeout=120) as c:
        r = await c.post(
            f"{cfg['base_url']}/chat/completions",
            headers={"Authorization": f"Bearer {key}"},
            json={"model": cfg["model"], "messages": messages, "temperature": temperature},
        )
        r.raise_for_status()
        return r.json()["choices"][0]["message"]["content"]
agents.py — orchestration multi-agents
python
from vector_store import search
from llm_router import chat

class Agent:
    name = "Agent"
    def __init__(self, provider: str):
        self.provider = provider

class PlannerAgent(Agent):
    name = "Planificateur"
    async def run(self, question: str) -> str:
        return await chat(self.provider, [
            {"role": "system", "content": "Tu décomposes une question d'élève en sous-tâches claires. Réponds en 3 puces max."},
            {"role": "user", "content": question},
        ])

class RetrieverAgent(Agent):
    name = "Chercheur"
    def run(self, question: str, k=6):
        return search(question, k=k)

class SolverAgent(Agent):
    name = "Solveur"
    async def run(self, question: str, context: str, plan: str) -> str:
        return await chat(self.provider, [
            {"role": "system", "content":
             "Tu es un professeur. Réponds en français, étape par étape, en citant les passages du cours fournis entre <cours>...</cours>. "
             "Si l'info manque, dis-le."},
            {"role": "user", "content": f"<cours>{context}</cours>\n\nPlan: {plan}\n\nQuestion: {question}"},
        ])

class TutorAgent(Agent):
    name = "Tuteur"
    async def run(self, question: str, answer: str) -> str:
        return await chat(self.provider, [
            {"role": "system", "content":
             "Reformule la réponse pour un élève, propose 2 questions de vérification et un exercice d'application."},
            {"role": "user", "content": f"Question: {question}\nRéponse brute: {answer}"},
        ])

async def run_pipeline(question: str, provider: str = "mistral"):
    events = []
    plan = await PlannerAgent(provider).run(question)
    events.append(("plan", plan))

    hits = RetrieverAgent(provider).run(question)
    context = "\n---\n".join(h["text"] for h in hits) or "(aucun cours pertinent)"
    events.append(("retrieval", [h["meta"] for h in hits]))

    answer = await SolverAgent(provider).run(question, context, plan)
    events.append(("answer", answer))

    tuto = await TutorAgent(provider).run(question, answer)
    events.append(("tutor", tuto))
    return events

from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from datetime import datetime
from database import SessionLocal, ApiKey
from security import encrypt, mask
from llm_router import verify_key

router = APIRouter(prefix="/api/keys")

class KeyIn(BaseModel):
    provider: str   # "mistral" | "agnes"
    key: str

@router.post("/save")
async def save(payload: KeyIn):
    if payload.provider not in ("mistral", "agnes"):
        raise HTTPException(400, "Fournisseur inconnu")

    ok = await verify_key(payload.provider, payload.key)
    if not ok:
        raise HTTPException(401, f"Clé {payload.provider} invalide ou non autorisée")

    db = SessionLocal()
    row = db.query(ApiKey).filter_by(provider=payload.provider).first()
    if row:
        row.encrypted = encrypt(payload.key)
        row.verified_at = datetime.utcnow()
    else:
        db.add(ApiKey(provider=payload.provider,
                      encrypted=encrypt(payload.key),
                      verified_at=datetime.utcnow()))
    db.commit(); db.close()

    return {
        "status": "ok",
        "provider": payload.provider,
        "masked": mask(payload.key),
        "message": f"✅ Clé {payload.provider} vérifiée et enregistrée."
    }

@router.get("/status")
def status():
    db = SessionLocal()
    rows = db.query(ApiKey).all()
    db.close()
    return {r.provider: {"masked": mask("xxxx"), "verified_at": r.verified_at} for r in rows}

import shutil, uuid
from config import UPLOAD_DIR

from vector_store import index_chunks

ALLOWED = {".pdf", ".docx", ".txt", ".png", ".jpg", ".jpeg"}

@router.post("/upload")
async def upload(file: UploadFile = File(...), title: str = Form(...), subject: str = Form("général")):
    ext = Path(file.filename).suffix.lower()
    if ext not in ALLOWED:
        return {"error": f"Format {ext} non supporté"}
    dest = UPLOAD_DIR / f"{uuid.uuid4().hex}{ext}"
    with dest.open("wb") as f:
        shutil.copyfileobj(file.file, f)

    text = extract_text(dest)
    chunks = chunk(text)

    db = SessionLocal()
    doc = Document(title=title, subject=subject, kind=ext[1:], path=str(dest), text=text[:200_000])
    db.add(doc); db.commit(); db.refresh(doc)
    index_chunks(doc.id, title, chunks)
    db.close()

    return {"id": doc.id, "title": title, "chunks": len(chunks),
            "message": f"📚 '{title}' importé ({len(chunks)} segments indexés)."}

7. Flux utilisateur complet (comme tu le décris)
Import → l'utilisateur glisse ses PDF/images/DOCX avec un titre → backend extrait texte + OCR → chunks indexés dans ChromaDB.

Agents → l'UI affiche l'état : "Planificateur prêt ✅ / Chercheur : 6 passages trouvés ✅ / Solveur prêt ✅ / Tuteur prêt ✅".

Clés API → l'utilisateur colle sa clé Mistral (et/ou Agnes) → POST /api/keys/save → vérification en direct auprès du provider → confirmation « ✅ Clé vérifiée et enregistrée (sk-…4f2a) ».

Questions → tirées des cours (RAG), de l'actualité (ajoute un WebSearchAgent via Tavily/Brave), ou de livres importés → même pipeline.

Réponses → étapes numérotées, citations des passages du cours, questions de vérification.

// lib/sse.ts
export async function sseStream(url: string, body: any, onEvent: (type: string, data: any) => void, onEnd: () => void) {
  const res = await fetch(url, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  });
  const reader = res.body!.getReader();
  const decoder = new TextDecoder();
  let buffer = "";
  
  while (true) {
    const { value, done } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });
    
    let idx;
    while ((idx = buffer.indexOf("\n\n")) !== -1) {
      const chunk = buffer.slice(0, idx);
      buffer = buffer.slice(idx + 2);
      
      let eventType = "message";
      let dataLines: string[] = [];
      for (const line of chunk.split("\n")) {
        if (line.startsWith("event:")) eventType = line.slice(6).trim();
        else if (line.startsWith("data:")) dataLines.push(line.slice(5).trim());
      }
      if (dataLines.length) {
        onEvent(eventType, JSON.parse(dataLines.join("\n")));
      }
    }
  }
  onEnd();
}

import asyncio
from typing import AsyncIterator, Callable, Any

class EventBus:
    """Bus d'événements asyncio pour pousser des messages SSE depuis les agents."""
    def __init__(self):
        self._queue: asyncio.Queue[dict] = asyncio.Queue()
        self._closed = False

    async def emit(self, event: str, data: dict):
        if self._closed:
            return
        await self._queue.put({"event": event, "data": data})

    async def close(self):
        if self._closed:
            return
        self._closed = True
        await self._queue.put({"event": "__end__", "data": {}})

    async def stream(self) -> AsyncIterator[dict]:
        while True:
            ev = await self._queue.get()
            if ev["event"] == "__end__":
                break
            yield ev

    @staticmethod
    def format_sse(event: str, data: dict) -> str:
        import json
        return f"event: {event}\ndata: {json.dumps(data, ensure_ascii=False)}\n\n"

class BaseAgent:
    id: str = "agent"
    label: str = "Agent"

    def __init__(self, provider: str, bus: EventBus):
        self.provider = provider
        self.bus = bus

    async def _start(self, **extra):
        await self.bus.emit("agent.start", {"id": self.id, "label": self.label, **extra})

    async def _done(self, summary: str, **payload):
        await self.bus.emit("agent.done", {"id": self.id, "label": self.label,
                                           "summary": summary, **payload})

    async def _log(self, message: str):
        await self.bus.emit("agent.log", {"id": self.id, "message": message})


class PlannerAgent(BaseAgent):
    id, label = "planner", "🧭 Planificateur"

    async def run(self, question: str) -> str:
        await self._start()
        await self._log("Décomposition de la question…")
        plan = await chat(self.provider, [
            {"role": "system", "content":
             "Tu es un planificateur pédagogique. Décompose la question en 3 sous-tâches "
             "maximum, en français, sous forme de puces très courtes."},
            {"role": "user", "content": question},
        ], temperature=0.2)
        await self._done("Plan établi", plan=plan)
        return plan


class RetrieverAgent(BaseAgent):
    id, label = "retriever", "🔍 Chercheur"

    async def run(self, question: str, k: int = 6):
        await self._start()
        await self._log(f"Recherche dans les cours (top {k})…")
        hits = search(question, k=k)
        if not hits:
            await self._log("Aucun passage pertinent trouvé dans les cours indexés.")
        else:
            await self._log(f"{len(hits)} passages retenus.")
        # Stream les passages un par un
        for i, h in enumerate(hits):
            await self.bus.emit("retrieval.hit", {
                "index": i,
                "title": h["meta"].get("title"),
                "score": round(h["score"], 3),
                "preview": h["text"][:180] + ("…" if len(h["text"]) > 180 else ""),
            })
        await self._done(f"{len(hits)} passages", hits=[h["meta"] for h in hits])
        return hits


class SolverAgent(BaseAgent):
    id, label = "solver", "✍️ Solveur"

    async def run(self, question: str, context: str, plan: str) -> str:
        await self._start()
        await self._log("Rédaction de la réponse étape par étape…")
        # Token par token si le provider le supporte ; sinon réponse d'un bloc
        answer = await chat(self.provider, [

    async for ev in bus.stream():
        yield ev
    await task

ts
export type SseHandler = (event: string, data: any) => void;

export async function sseStream(
  url: string,
  body: any,
  onEvent: SseHandler,
  signal?: AbortSignal,
) {
  const res = await fetch(url, {
    method: "POST",
    headers: { "Content-Type": "application/json", Accept: "text/event-stream" },
    body: JSON.stringify(body),
    signal,
  });
  if (!res.ok || !res.body) throw new Error(`SSE HTTP ${res.status}`);

  const reader = res.body.getReader();
  const decoder = new TextDecoder("utf-8");
  let buffer = "";

  while (true) {
    const { value, done } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });

    let idx: number;
    while ((idx = buffer.indexOf("\n\n")) !== -1) {
      const raw = buffer.slice(0, idx);
      buffer = buffer.slice(idx + 2);

      let event = "message";
      const dataLines: string[] = [];
      for (const line of raw.split("\n")) {
        if (line.startsWith(":")) continue;            // keep-alive
        if (line.startsWith("event:")) event = line.slice(6).trim();
        else if (line.startsWith("data:")) dataLines.push(line.slice(5).trim());
      }
      if (!dataLines.length) continue;
      try {
        onEvent(event, JSON.parse(dataLines.join("\n")));
      } catch {
        // ignore
      }
    }
  }
}

const API = process.env.NEXT_PUBLIC_API_URL ?? "http://localhost:8000";

const AGENT_ORDER: { id: string; label: string }[] = [
  { id: "planner", label: "🧭 Planificateur" },
  { id: "retriever", label: "🔍 Chercheur" },
  { id: "solver", label: "✍️ Solveur" },
  { id: "tutor", label: "👨‍🏫 Tuteur" },
];

function freshAgents(): AgentState[] {
  return AGENT_ORDER.map((a) => ({ ...a, status: "idle", logs: [] }));
}

export function ChatStream() {
  const [question, setQuestion] = useState("");
  const [provider, setProvider] = useState<"mistral" | "agnes">("mistral");
  const [agents, setAgents] = useState<AgentState[]>(freshAgents());
  const [answer, setAnswer] = useState("");
  const [tutor, setTutor] = useState("");
  const [busy, setBusy] = useState(false);
  const [hits, setHits] = useState<any[]>([]);
  const abortRef = useRef<AbortController | null>(null);

  function updateAgent(id: string, patch: Partial<AgentState> | ((a: AgentState) => AgentState)) {
    setAgents((prev) => prev.map((a) => {
      if (a.id !== id) return a;
      return typeof patch === "function" ? patch(a) : { ...a, ...patch };
    }));
  }

  async function ask() {
    if (!question.trim()) return;
    setBusy(true);
    setAnswer("");
    setTutor("");
    setHits([]);
    setAgents(freshAgents());

    const ctrl = new AbortController();
    abortRef.current = ctrl;

    try {
      await sseStream(
        `${API}/api/chat/ask/stream`,
        { question: question.trim(), provider },
        (event, data) => {
          switch (event) {
            case "pipeline.start":
              toast.info("Pipeline démarré");
              break;

            case "agent.start":
              updateAgent(data.id, { status: "running" });
              break;

            case "agent.log":
              updateAgent(data.id, (a) => ({ ...a, logs: [...a.logs, data.message] }));
              break;

            case "agent.done":
              updateAgent(data.id, { status: "done", summary: data.summary });
              break;

            case "retrieval.hit":
              setHits((h) => [...h, data]);
              break;

            case "answer.token":
              setAnswer((a) => a + data.token);
              break;

            case "pipeline.done":
              if (data.answer) setAnswer(data.answer);
              toast.success("Réponse prête ✅");
              break;

            case "pipeline.error":
              toast.error(data.message);
              setAgents((prev) => prev.map((a) =>
                a.status === "running" ? { ...a, status: "error" } : a));
              break;
          }
        },
        ctrl.signal,
      );
    } catch (e: any) {
      if (e.name !== "AbortError") toast.error(e.message ?? "Erreur de streaming");
    } finally {
      setBusy(false);
      abortRef.current = null;
    }
  }

  function stop() {
    abortRef.current?.abort();
    setBusy(false);
    setAgents((prev) => prev.map((a) =>
      a.status === "running" ? { ...a, status: "error", summary: "interrompu" } : a));
  }

  return (
    <div className="grid gap-6 md:grid-cols-[1fr_380px]">
      {/* Colonne gauche : question + réponse */}
      <div className="space-y-4">
        <Card>
          <CardHeader><CardTitle>💬 Poser une question</CardTitle></CardHeader>
          <CardContent className="space-y-3">
            <Textarea
              rows={3}
              value={question}
              onChange={(e) => setQuestion(e.target.value)}
              placeholder="Ex : Explique la dérivée d'une fonction composée, avec un exemple du cours."
            />
            <div className="flex gap-2">
              <Select value={provider} onValueChange={(v) => setProvider(v as any)}>
                <SelectTrigger className="w-40"><SelectValue /></SelectTrigger>
                <SelectContent>
                  <SelectItem value="mistral">Mistral</SelectItem>
                  <SelectItem value="agnes">Agnes</SelectItem>
                </SelectContent>
              </Select>
              <Button onClick={ask} disabled={busy} className="flex-1">
                <Send className="mr-2 h-4 w-4" />
                {busy ? "Génération…" : "Envoyer"}
              </Button>
              {busy && (
                <Button variant="outline" onClick={stop}>
                  <Square className="h-4 w-4" />
                </Button>
              )}
            </div>
          </CardContent>
        </Card>

        {(answer || tutor) && (
          <Card>
            <CardHeader><CardTitle>Réponse</CardTitle></CardHeader>
            <CardContent className="space-y-4">

python
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from datetime import datetime
from database import SessionLocal, ApiKey
from security import encrypt, decrypt, mask
from providers.registry import list_providers, get_provider

router = APIRouter(prefix="/api/keys")


class KeyIn(BaseModel):
    provider: str
    key: str


# ---------- Liste des providers disponibles ----------
@router.get("/providers")
def providers():
    return {"providers": list_providers()}


# ---------- TEST seul : ne sauvegarde rien ----------
@router.post("/test")
async def test_key(payload: KeyIn):
    if payload.provider not in list_providers():
        raise HTTPException(400, f"Fournisseur inconnu : {payload.provider}")
    ok, msg = await get_provider(payload.provider).verify(payload.key)
    return {
        "ok": ok,
        "provider": payload.provider,
        "message": msg,
        "masked": mask(payload.key),
    }


# ---------- TEST + SAVE ----------
@router.post("/save")
async def save(payload: KeyIn):
    if payload.provider not in list_providers():
        raise HTTPException(400, "Fournisseur inconnu")

    ok, msg = await get_provider(payload.provider).verify(payload.key)
    if not ok:
        raise HTTPException(401, msg)

    db = SessionLocal()
    row = db.query(ApiKey).filter_by(provider=payload.provider).first()
    now = datetime.utcnow()
    if row:
        row.encrypted = encrypt(payload.key)
        row.verified_at = now
        action = "mise à jour"
    else:
        db.add(ApiKey(provider=payload.provider,
                      encrypted=encrypt(payload.key),
                      verified_at=now))
        action = "enregistrée"
    db.commit()
    db.close()

    return {
        "status": "ok",
        "provider": payload.provider,
        "masked": mask(payload.key),
        "verified_at": now.isoformat(),
        "message": f"✅ Clé {payload.provider} {action} ({mask(payload.key)}).",
    }


# ---------- Statut détaillé ----------
@router.get("/status")
def status():
    db = SessionLocal()
    rows = db.query(ApiKey).all()
    db.close()
    return {
        r.provider: {
            "configured": True,
            "masked": mask(decrypt(r.encrypted)),
            "verified_at": r.verified_at.isoformat() if r.verified_at else None,
        }
        for r in rows
    }


# ---------- Suppression ----------
@router.delete("/{provider}")
def delete_key(provider: str):
    db = SessionLocal()
    row = db.query(ApiKey).filter_by(provider=provider).first()
    if not row:
        db.close()
        raise HTTPException(404, f"Aucune clé {provider} enregistrée")
    db.delete(row)
    db.commit()
    db.close()
    return {"status": "ok", "message": f"🗑️ Clé {provider} supprimée."}

export type Provider = "mistral" | "agnes";

export type KeyStatus = {
  configured: boolean;
  masked?: string;
  verified_at?: string;
};

export const api = {
  // ... existant ...

  async providers(): Promise<{ providers: Provider[] }> {
    const r = await fetch(`${API}/api/keys/providers`);
    return r.json();
  },

  async testKey(provider: Provider, key: string) {
    const r = await fetch(`${API}/api/keys/test`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ provider, key }),
    });
    const j = await r.json();
    if (!r.ok) throw new Error(j.detail ?? "Erreur");
    return j as { ok: boolean; provider: Provider; message: string; masked: string };
  },

  async saveKey(provider: Provider, key: string) {
    const r = await fetch(`${API}/api/keys/save`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ provider, key }),
    });
    const j = await r.json();
    if (!r.ok) throw new Error(j.detail ?? "Erreur");
    return j as { message: string; masked: string; verified_at: string };
  },

  async keysStatus(): Promise<Record<string, KeyStatus>> {
    const r = await fetch(`${API}/api/keys/status`);
    return r.json();
  },

  async deleteKey(provider: Provider) {
    const r = await fetch(`${API}/api/keys/${provider}`, { method: "DELETE" });
    const j = await r.json();
    if (!r.ok) throw new Error(j.detail ?? "Erreur");
    return j;
  },
};

  mistral: {
    label: "Mistral",
    hint: "Clé de type sk-... (console.mistral.ai)",
    docs: "https://console.mistral.ai/api-keys/",
  },
  agnes: {
    label: "Agnes",
    hint: "Clé fournie par Agnes AI",
    docs: "#",
  },

export function ApiKeyForm() {
  const [providers, setProviders] = useState<Provider[]>(["mistral", "agnes"]);
  const [provider, setProvider] = useState<Provider>("mistral");
  const [key, setKey] = useState("");
  const [showKey, setShowKey] = useState(false);
  const [status, setStatus] = useState<Record<string, KeyStatus>>({});

  const [testing, setTesting] = useState(false);
  const [saving, setSaving] = useState(false);
  const [testResult, setTestResult] = useState<{ ok: boolean; message: string } | null>(null);

  // Charger providers + statut au montage
  useEffect(() => {
    (async () => {
      try {
        const { providers } = await api.providers();
        setProviders(providers);
      } catch { /* fallback */ }
      refreshStatus();
    })();
  }, []);

  // Reset le résultat de test quand on change de provider ou de clé
  useEffect(() => { setTestResult(null); }, [provider, key]);

  async function refreshStatus() {
    try { setStatus(await api.keysStatus()); } catch { /* ignore */ }
  }

  async function handleTest() {
    if (!key.trim()) return toast.error("Colle une clé à tester.");
    setTesting(true);
    setTestResult(null);
    try {
      const r = await api.testKey(provider, key.trim());
      setTestResult({ ok: r.ok, message: r.message });
      if (r.ok) toast.success(r.message);
      else toast.error(r.message);
    } catch (e: any) {
      setTestResult({ ok: false, message: e.message });
      toast.error(e.message);
    } finally {
      setTesting(false);
    }
  }

  async function handleSave() {
    if (!key.trim()) return toast.error("Colle une clé avant d'enregistrer.");
    setSaving(true);
    try {
      const r = await api.saveKey(provider, key.trim());
      toast.success(r.message);
      setKey("");
      setTestResult(null);
      await refreshStatus();
    } catch (e: any) {
      toast.error(e.message);
    } finally {
      setSaving(false);
    }
  }

  async function handleDelete(p: Provider) {
    try {
      const r = await api.deleteKey(p);
      toast.success(r.message);
      await refreshStatus();
    } catch (e: any) {
      toast.error(e.message);
    }
  }

  const current = status[provider];

  return (
    <TooltipProvider delayDuration={200}>
      <div className="space-y-6">

        {/* ---------- Carte principale : formulaire ---------- */}

                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  {providers.map((p) => (
                    <SelectItem key={p} value={p}>
                      <div className="flex items-center gap-2">
                        <span className="capitalize">{PROVIDER_META[p]?.label ?? p}</span>
                        {status[p]?.configured && (
                          <Badge variant="secondary" className="h-5 px-1.5 text-[10px]">
                            configurée
                          </Badge>
                        )}
                      </div>
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
              <p className="text-xs text-muted-foreground">
                {PROVIDER_META[provider]?.hint}
              </p>
            </div>

            {/* Saisie de la clé */}
            <div className="grid gap-2">
              <label className="text-sm font-medium">Clé secrète</label>
              <div className="flex gap-2">
                <div className="relative flex-1">
                  <KeyRound className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />

                    type={showKey ? "text" : "password"}
                    placeholder="sk-..."
                    value={key}
                    onChange={(e) => setKey(e.target.value)}
                    onKeyDown={(e) => e.key === "Enter" && handleTest()}
                    className="pl-9 pr-10 font-mono text-sm"
                    autoComplete="off"
                    spellCheck={false}
                  />
                  <button
                    type="button"
                    onClick={() => setShowKey((s) => !s)}

                  </button>
                </div>
              </div>
            </div>

            {/* Résultat du test */}
            {testResult && (
              <div
                className={
                  "flex items-start gap-2 rounded-md border p-3 text-sm " +
                  (testResult.ok
                    ? "border-emerald-200 bg-emerald-50 text-emerald-800 dark:border-emerald-800 dark:bg-emerald-950/40 dark:text-emerald-300"
                    : "border-red-200 bg-red-50 text-red-800 dark:border-red-800 dark:bg-red-950/40 dark:text-red-300")
                }
              >
                {testResult.ok
                  ? <CheckCircle2 className="mt-0.5 h-4 w-4 shrink-0" />
                  : <XCircle className="mt-0.5 h-4 w-4 shrink-0" />}

            {/* Actions */}

              const configured = !!st?.configured;
              return (

                  {i > 0 && <Separator className="my-3" />}
